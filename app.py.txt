const express = require('express');

const app = express();
const PORT = 3000;

app.get('/', (req, res) => {
    res.send('Node.js application deployed successfully!');
});

app.get('/health', (req, res) => {
    res.json({
        status: 'UP',
        application: 'Node.js App'
    });
});

app.listen(PORT, '0.0.0.0', () => {
    console.log(`Server running on port ${PORT}`);
});
